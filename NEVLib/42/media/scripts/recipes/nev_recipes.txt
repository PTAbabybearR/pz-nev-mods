        {
            item 1 tags[base:screwdriver] mode:keep,
            item 2 [Base.ElectricWire],
            item 4 [Base.ElectronicsScrap],
            item 1 [Base.MetalPipe],
        }
        outputs
        {
            item 1 NEVLib.HomeCharger2,
        }
    }
